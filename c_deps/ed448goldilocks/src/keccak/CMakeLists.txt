# Pinned XKCP low-level implementations; compile ISA-specific objects separately.
option(ENABLE_KECCAK_OPTIMIZATIONS "Enable runtime-selected Keccak CPU backends." ON)
set(XKCP_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/xkcp/lib)
set(XKCP_COMMON ${XKCP_ROOT}/common ${XKCP_ROOT}/low/common ${XKCP_ROOT}/low/KeccakP-1600/common)
set(XKCP_PLAIN ${XKCP_ROOT}/low/KeccakP-1600/plain-64bits)
set(XKCP_INCLUDES ${XKCP_COMMON} ${XKCP_PLAIN} ${CMAKE_CURRENT_BINARY_DIR})
set(KECCAK_OBJECTS)

function(decaf_keccak_object name source)
    add_library(${name} OBJECT ${source})
    target_include_directories(${name} PRIVATE ${XKCP_INCLUDES} ${ARGN})
    set_target_properties(${name} PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
    set(KECCAK_OBJECTS ${KECCAK_OBJECTS} $<TARGET_OBJECTS:${name}> PARENT_SCOPE)
endfunction()

decaf_keccak_object(decaf-keccak-plain ${XKCP_PLAIN}/KeccakP-1600-opt64.c ${XKCP_PLAIN}/SnP)

if(ENABLE_KECCAK_OPTIMIZATIONS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
    include(CheckCCompilerFlag)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        check_c_compiler_flag("-march=armv8.2-a+sha3" DECAF_CAN_BUILD_SHA3)
        if(DECAF_CAN_BUILD_SHA3)
            enable_language(ASM)
            set(DECAF_KECCAK_SHA3 1)
            decaf_keccak_object(decaf-keccak-sha3
                "${XKCP_ROOT}/low/KeccakP-1600/ARMv8A-SHA3/KeccakP-1600-x1-v84a-asm.S;${XKCP_ROOT}/low/KeccakP-1600-times2/ARMv8A/KeccakP-1600-times2-x2-v84a-asm.S")
            target_compile_options(decaf-keccak-sha3 PRIVATE -march=armv8.2-a+sha3)
            decaf_keccak_object(decaf-keccak-k12 ${CMAKE_CURRENT_SOURCE_DIR}/k12/KeccakP-1600-ARMv8Asha3.S)
            target_compile_options(decaf-keccak-k12 PRIVATE -march=armv8.2-a+sha3)
            if(APPLE)
                target_compile_options(decaf-keccak-k12 PRIVATE -Wa,-defsym,old_gas_syntax=1)
            endif()
        endif()
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$" AND NOT WIN32)
        enable_language(ASM)
        set(DECAF_KECCAK_X86 1)
        set(XKCP_SINGLE ${XKCP_ROOT}/low/KeccakP-1600)
        decaf_keccak_object(decaf-keccak-x86
            "${XKCP_SINGLE}/AVX2/KeccakP-1600-AVX2.s;${XKCP_SINGLE}/AVX512/KeccakP-1600-AVX512.s")
        if(APPLE)
            target_compile_options(decaf-keccak-x86 PRIVATE -Wa,-defsym,old_gas_syntax=1)
        endif()
        set(XKCP_TWO ${XKCP_ROOT}/low/KeccakP-1600-times2/SIMD128)
        set(XKCP_FOUR ${XKCP_ROOT}/low/KeccakP-1600-times4/AVX2)
        set(XKCP_EIGHT ${XKCP_ROOT}/low/KeccakP-1600-times8/AVX512)
        decaf_keccak_object(decaf-keccak-ssse3 ${XKCP_TWO}/KeccakP-1600-times2-SIMD128.c ${XKCP_TWO})
        target_compile_options(decaf-keccak-ssse3 PRIVATE -mssse3)
        decaf_keccak_object(decaf-keccak-avx2 ${XKCP_FOUR}/KeccakP-1600-times4-AVX2.c ${XKCP_FOUR} ${XKCP_FOUR}/SnP)
        target_compile_options(decaf-keccak-avx2 PRIVATE -mavx2)
        decaf_keccak_object(decaf-keccak-avx512 ${XKCP_EIGHT}/KeccakP-1600-times8-AVX512.c ${XKCP_EIGHT})
        target_compile_options(decaf-keccak-avx512 PRIVATE -mavx512f -mavx512vl -mavx512bw -mavx512dq)
        list(APPEND XKCP_INCLUDES ${XKCP_SINGLE}/AVX2 ${XKCP_SINGLE}/AVX512 ${XKCP_TWO} ${XKCP_FOUR} ${XKCP_EIGHT})
    endif()
endif()
configure_file(config.h.in config.h)
add_library(decaf-keccak-adapter OBJECT keccak.c batch.c)
add_dependencies(decaf-keccak-adapter decaf-generated-code)
target_include_directories(decaf-keccak-adapter PRIVATE ${XKCP_INCLUDES})
set_target_properties(decaf-keccak-adapter PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(decaf-keccak-adapter PRIVATE -std=c11 -fno-strict-aliasing)
endif()
set(DECAF_KECCAK_OBJECTS ${KECCAK_OBJECTS} $<TARGET_OBJECTS:decaf-keccak-adapter> PARENT_SCOPE)
